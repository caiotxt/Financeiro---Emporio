import pandas as pd
import numpy as np
import argparse
import os



def clean_sku(x):
    if pd.isna(x):
        return np.nan
    s = str(x).strip()
    return s if s else np.nan

def ensure_dir(path):
    os.makedirs(path, exist_ok=True)

def main():
    parser = argparse.ArgumentParser(
        description="DRE Mercado Livre com lucro por SKU"
    )
    parser.add_argument("--collection", required=True, help="Arquivo collection do Mercado Livre (.xlsx)")
    parser.add_argument("--dimi", required=True, help="Planilha do Dimi (.xlsx)")
    parser.add_argument("--dimi-sheet", default="PRODUTOS", help="Aba de produtos da Planilha do Dimi")
    parser.add_argument("--outdir", default="saida", help="Pasta de saída")

    args = parser.parse_args()

   
    df = pd.read_excel(
        args.collection,
        engine="openpyxl",
        engine_kwargs={"read_only": True, "data_only": True}
    )
    dimi = pd.read_excel(args.dimi, sheet_name=args.dimi_sheet)

    # Normalizando a parada
    df["SKU"] = df["SKU do produto (seller_custom_field)"].apply(clean_sku)

    df["net_received_amount"] = pd.to_numeric(
        df["Valor total recebido (net_received_amount)"], errors="coerce"
    ).fillna(0)

    df["transaction_amount"] = pd.to_numeric(
        df["Valor do produto (transaction_amount)"], errors="coerce"
    ).fillna(0)

   
    df["qtd"] = np.where(
        df["transaction_amount"] > 0, 1,
        np.where(df["transaction_amount"] < 0, -1, 0)
    )

    sku_qty = (
        df.dropna(subset=["SKU"])
          .groupby("SKU", as_index=False)["qtd"]
          .sum()
          .query("qtd != 0")
    )

   
    receita_sku = (
        df.dropna(subset=["SKU"])
          .groupby("SKU", as_index=False)["net_received_amount"]
          .sum()
          .rename(columns={"net_received_amount": "receita_liquida_sku"})
    )

    
    dimi = dimi.rename(columns={dimi.columns[0]: "SKU"})
    dimi["SKU"] = dimi["SKU"].apply(clean_sku)

    custo_col = [c for c in dimi.columns if c.lower() == "custo"][0]
    dimi[custo_col] = pd.to_numeric(dimi[custo_col], errors="coerce")

    dimi_cost = (
        dimi[["SKU", custo_col]]
        .dropna(subset=["SKU"])
        .drop_duplicates(subset=["SKU"], keep="last")
        .rename(columns={custo_col: "custo_unit"})
    )

    
    resultado_sku = (
        sku_qty
        .merge(receita_sku, on="SKU", how="left")
        .merge(dimi_cost, on="SKU", how="left")
    )

    resultado_sku["cmv_total_sku"] = resultado_sku["qtd"] * resultado_sku["custo_unit"]
    resultado_sku["lucro_sku"] = resultado_sku["receita_liquida_sku"] - resultado_sku["cmv_total_sku"]
    resultado_sku["margem_sku"] = np.where(
        resultado_sku["receita_liquida_sku"] != 0,
        resultado_sku["lucro_sku"] / resultado_sku["receita_liquida_sku"],
        np.nan
    )

    
    RECEITA_LIQUIDA = resultado_sku["receita_liquida_sku"].sum()
    CMV_TOTAL = resultado_sku["cmv_total_sku"].sum()
    LUCRO = resultado_sku["lucro_sku"].sum()
    MARGEM = LUCRO / RECEITA_LIQUIDA if RECEITA_LIQUIDA else np.nan

    
    resumo = pd.DataFrame([{
        "Receita Líquida": RECEITA_LIQUIDA,
        "CMV": CMV_TOTAL,
        "Lucro Líquido": LUCRO,
        "Margem": MARGEM
    }])

    
    ensure_dir(args.outdir)
    out_path = os.path.join(args.outdir, "DRE_Mercado_Livre.xlsx")

    with pd.ExcelWriter(out_path, engine="openpyxl") as writer:
        resultado_sku.sort_values("lucro_sku", ascending=False).to_excel(
            writer, index=False, sheet_name="Lucro_por_SKU"
        )
        resumo.to_excel(writer, index=False, sheet_name="Resumo_DRE")

    print(f"✅ Relatório gerado: {out_path}")
    print(f"Lucro Total: {LUCRO:,.2f}")

if __name__ == "__main__":
    main()
